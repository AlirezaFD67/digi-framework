# 📝 Article Endpoint

Complete documentation for Article (Learning) endpoint that manages articles, categories, authors, and content.

## 🎯 Overview

The Article endpoint provides a complete solution for managing learning content including:

- **Article Management** - Create, read, update, and publish articles
- **Category Management** - Organize articles by categories
- **Author Management** - Manage article authors
- **Content Management** - Rich content with blocks
- **File Management** - Upload and manage article files
- **Publication Control** - Accept/reject article publication

### Create Article Mutation

```typescript
import { useAddAdminLearningList } from "@workspace/framework";

function CreateArticleForm() {
  const createArticle = useAddAdminLearningList();

  const handleSubmit = async (formData) => {
    try {
      const result = await createArticle.mutateAsync({
        catID: formData.categoryId,
        itmTitle: formData.title,
        itmSlug: formData.slug
      });
      
      console.log("Article created:", result);
    } catch (error) {
      console.error("Failed to create article:", error);
    }
  };

  return (
    <form onSubmit={handleSubmit}>
      {/* Form fields */}
      <button 
        type="submit" 
        disabled={createArticle.isPending}
      >
        {createArticle.isPending ? "Creating..." : "Create Article"}
      </button>
    </form>
  );
}
```

### Accept/Reject Article Mutation

```typescript
import { useAcceptLearningMutation } from "@workspace/framework";

function ArticleActions({ articleId, isAccepted }: Props) {
  const acceptMutation = useAcceptLearningMutation();

  const handleToggleStatus = async () => {
    try {
      await acceptMutation.mutateAsync({
        itmID: articleId,
        isAccepted: isAccepted ? 0 : 1
      });
      
      alert("Article status updated!");
    } catch (error) {
      console.error("Failed to update status:", error);
    }
  };

  return (
    <button 
      onClick={handleToggleStatus}
      disabled={acceptMutation.isPending}
    >
      {isAccepted ? "Unpublish" : "Publish"}
    </button>
  );
}
```

## 🎨 Complete Example - Article Management Page

```tsx
"use client";

import { useState, useEffect, useCallback } from "react";
import { useRouter } from "next/navigation";
import {
  useAdminLearningListQuery,
  useAcceptLearningMutation,
  useAdminCategoryListQuery
} from "@workspace/framework";

export default function ArticleManagementPage() {
  const router = useRouter();
  const [isAccepted, setIsAccepted] = useState<number>(1);
  const [currentPage, setCurrentPage] = useState<number>(1);
  const [searchTerm, setSearchTerm] = useState<string>("");

  // Queries
  const { data, isLoading, error } = useAdminLearningListQuery({
    isAccepted,
    pageNo: currentPage,
    rowCount: 10,
    itmTitle: searchTerm || undefined
  });

  const { data: categories } = useAdminCategoryListQuery();

  // Mutations
  const acceptMutation = useAcceptLearningMutation();

  // Handlers
  const handleAcceptLearning = useCallback(
    async (itmID: string, newStatus: number) => {
      try {
        await acceptMutation.mutateAsync({ 
          itmID, 
          isAccepted: newStatus 
        });
      } catch (error) {
        console.error("Failed to update article status:", error);
      }
    },
    [acceptMutation]
  );

  const handleEdit = useCallback(
    (id: string) => {
      router.push(`/article/edit/${id}`);
    },
    [router]
  );

  const handlePageChange = useCallback((page: number) => {
    setCurrentPage(page);
  }, []);

  const handleTabChange = useCallback((newIsAccepted: number) => {
    setIsAccepted(newIsAccepted);
    setCurrentPage(1);
  }, []);

  // Reset page when search changes
  useEffect(() => {
    setCurrentPage(1);
  }, [searchTerm]);

  if (isLoading) {
    return (
      <div className="flex items-center justify-center h-64">
        <div className="text-lg">Loading articles...</div>
      </div>
    );
  }

  if (error) {
    return (
      <div className="text-red-500">
        Error loading articles: {error.message}
      </div>
    );
  }

  return (
    <div className="w-full space-y-6">
      {/* Header */}
      <div className="flex items-center justify-between">
        <h1 className="text-2xl font-bold">Article Management</h1>
        <button
          onClick={() => router.push("/article/create")}
          className="px-4 py-2 bg-blue-600 text-white rounded-lg"
        >
          Create New Article
        </button>
      </div>

      {/* Search */}
      <div className="flex gap-4">
        <input
          type="text"
          placeholder="Search articles..."
          value={searchTerm}
          onChange={(e) => setSearchTerm(e.target.value)}
          className="flex-1 px-4 py-2 border rounded-lg"
        />
      </div>

      {/* Tabs */}
      <div className="flex gap-2">
        <button
          onClick={() => handleTabChange(1)}
          className={`px-4 py-2 rounded-lg ${
            isAccepted === 1
              ? "bg-blue-600 text-white"
              : "bg-gray-200"
          }`}
        >
          Published Articles
        </button>
        <button
          onClick={() => handleTabChange(0)}
          className={`px-4 py-2 rounded-lg ${
            isAccepted === 0
              ? "bg-blue-600 text-white"
              : "bg-gray-200"
          }`}
        >
          Pending Review
        </button>
      </div>

      {/* Articles List */}
      <div className="grid gap-4">
        {data?.data?.entries?.map((article) => (
          <div
            key={article.itm_ID}
            className="border rounded-lg p-4 hover:shadow-lg transition"
          >
            <div className="flex items-start justify-between">
              <div className="flex-1">
                <h3 className="text-xl font-semibold">
                  {article.itm_Title}
                </h3>
                <p className="text-gray-600 mt-2">
                  {article.itm_Minimal}
                </p>
                <div className="flex items-center gap-4 mt-4 text-sm text-gray-500">
                  <span>Category: {article.itm_Cat}</span>
                  <span>Views: {article.itm_View}</span>
                  <span>
                    {new Date(article.DateOfCreation).toLocaleDateString()}
                  </span>
                </div>
              </div>

              <div className="flex gap-2">
                <button
                  onClick={() => handleEdit(article.itm_ID)}
                  className="px-3 py-1 bg-gray-200 rounded hover:bg-gray-300"
                >
                  Edit
                </button>
                <button
                  onClick={() =>
                    handleAcceptLearning(
                      article.itm_ID,
                      article.itm_Acc === 1 ? 0 : 1
                    )
                  }
                  disabled={acceptMutation.isPending}
                  className={`px-3 py-1 rounded ${
                    article.itm_Acc === 1
                      ? "bg-red-500 text-white hover:bg-red-600"
                      : "bg-green-500 text-white hover:bg-green-600"
                  }`}
                >
                  {article.itm_Acc === 1 ? "Unpublish" : "Publish"}
                </button>
              </div>
            </div>
          </div>
        ))}
      </div>

      {/* Pagination */}
      {data?.data?.metadata && (
        <div className="flex items-center justify-center gap-2 mt-6">
          <button
            onClick={() => handlePageChange(currentPage - 1)}
            disabled={currentPage === 1}
            className="px-4 py-2 border rounded disabled:opacity-50"
          >
            Previous
          </button>
          <span className="px-4">
            Page {currentPage} of {data.data.metadata.pageCount}
          </span>
          <button
            onClick={() => handlePageChange(currentPage + 1)}
            disabled={currentPage >= data.data.metadata.pageCount}
            className="px-4 py-2 border rounded disabled:opacity-50"
          >
            Next
          </button>
        </div>
      )}
    </div>
  );
}
```

## 🎯 Best Practices

### 1. **Always Use Query Keys**

```typescript
// ✅ Good
const { data } = useAdminLearningListQuery({
  isAccepted: 1,
  pageNo: 1,
  rowCount: 10
});

// ❌ Bad - Missing parameters in query key
```

### 2. **Handle Loading and Error States**

```typescript
const { data, isLoading, error } = useAdminLearningListQuery(params);

if (isLoading) return <LoadingSpinner />;
if (error) return <ErrorMessage error={error} />;
if (!data) return <EmptyState />;
```

### 3. **Use Optimistic Updates for Better UX**

```typescript
const acceptMutation = useAcceptLearningMutation();

// The mutation automatically invalidates cache
// and refetches the list
```

### 4. **Validate Before Mutations**

```typescript
const handleCreateArticle = async (formData) => {
  // Validate
  if (!formData.title || !formData.slug) {
    alert("Please fill all required fields");
    return;
  }

  try {
    await createArticle.mutateAsync(formData);
  } catch (error) {
    console.error(error);
  }
};
```
